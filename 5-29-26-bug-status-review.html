-- Migration 023: Add UPDATE policy to push_subscriptions for upsert support.
-- Bug: INSERT policy exists (015) but on_conflict upsert hits UPDATE path → 403.
BEGIN;
CREATE POLICY "player update push_subscriptions" ON push_subscriptions
  FOR UPDATE TO authenticated
  USING (
    tournament_id IN (
      SELECT tournament_id FROM players WHERE email = auth.email()
    )
  )
  WITH CHECK (
    tournament_id IN (
      SELECT tournament_id FROM players WHERE email = auth.email()
    )
  );
COMMIT;
