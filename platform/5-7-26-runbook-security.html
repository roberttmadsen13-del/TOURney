CREATE POLICY "owner or admin manage invitations"
  ON invitations FOR ALL TO authenticated

    EXISTS (
      SELECT 1 FROM tournaments t
      WHERE t.id = invitations.tournament_id
        AND (
          t.owner_email = auth.email()
          OR EXISTS (
            SELECT 1 FROM tournament_admins ta
            WHERE ta.tournament_id = t.id
              AND ta.player_id = auth.uid()
          )
        )
    )
  )

    EXISTS (
      SELECT 1 FROM tournaments t
      WHERE t.id = invitations.tournament_id
        AND (
          t.owner_email = auth.email()
          OR EXISTS (
            SELECT 1 FROM tournament_admins ta
            WHERE ta.tournament_id = t.id

import { createClient } from 'https://esm.sh/@supabase/supabase-js@2?target=deno';

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'apikey, content-type',
};

function json(data: unknown, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { ...corsHeaders, 'Content-Type': 'application/json' },
  });
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response(null, { headers: corsHeaders });

  // No JWT required â€” access_token IS the auth proof
  const { invitation_id, access_token } = await req.json();

  if (!invitation_id || !access_token) {
    return json({ error: 'Missing invitation_id or access_token' }, 400);
  }

  // Use service role to bypass RLS â€” but validate token first
  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  );

  const { data: inv, error: fetchErr } = await supabase
    .from('invitations')
    .select('id, access_token, status, email')
    .eq('id', invitation_id)
    .single();

  if (fetchErr || !inv) return json({ error: 'Invitation not found' }, 404);
  if (inv.access_token !== access_token) return json({ error: 'Invalid token' }, 403);

  // Idempotent â€” already accepted/declined, return current status
  if (inv.status !== 'pending') return json({ status: inv.status });

  const { error: updateErr } = await supabase
    .from('invitations')
    .update({ status: 'accepted' })
    .eq('id', invitation_id);

  if (updateErr) return json({ error: 'Update failed' }, 500);

  return json({ status: 'accepted', email: inv.email });
});

function safeUrl(u) {
